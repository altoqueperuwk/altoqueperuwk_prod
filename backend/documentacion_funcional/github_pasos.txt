# 1. Ver estado
git status

# 2. Agregar cambios
git add -A

# 3. Commitear
git commit -m "Descripción del cambio"

# 4. Subir a develop (lanza build en Cloudflare Preview)
git push

# 5. Cuando quieras lanzar a producción:
git checkout main
git merge develop
git push
git checkout develop
