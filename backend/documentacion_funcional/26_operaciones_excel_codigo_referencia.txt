function normalizeHeaderText(value: unknown): string {
  return String(value ?? "")
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/[^a-zA-Z0-9 ]/g, " ")
    .replace(/\s+/g, " ")
    .trim()
    .toLowerCase();
}

export type HeaderMatch = {
  headerRowIndex: number;
  idxDni: number;
  idxMonto: number;
  idxCodigoOperacion: number;
  idxFechaPago: number;
  idxHoraProceso: number;
  idxBancoIngreso: number;
  idxEstado: number;
};

function findHeaderIndexInRow(headers: string[], aliases: string[] | undefined): number {
  if (!aliases) return -1;

  const normalizedHeaders = headers.map(normalizeHeaderText);

  for (const alias of aliases) {
    const idx = normalizedHeaders.indexOf(normalizeHeaderText(alias));
    if (idx >= 0) return idx;
  }

  return -1;
}

export function findHeaderMatch(
  matrix: (string | number | null)[][],
  config: HeaderAliasConfig,
): HeaderMatch | null {
  for (let i = 0; i < matrix.length; i++) {
    const row = matrix[i] ?? [];
    const headers = row.map(String);

    const foundDni = findHeaderIndexInRow(headers, config.dni);
    const foundMonto = findHeaderIndexInRow(headers, config.monto);
    const foundCodigo = findHeaderIndexInRow(headers, config.codigo_operacion);
    const foundFecha = findHeaderIndexInRow(headers, config.fecha_pago);
    const foundHora = findHeaderIndexInRow(headers, config.hora_proceso);
    const foundBanco = findHeaderIndexInRow(headers, config.nombre_cliente_banco_ingreso);
    const foundEstado = findHeaderIndexInRow(headers, config.estado);

    // Regla minima vigente: monto + codigo_operacion.
    if (foundMonto >= 0 && foundCodigo >= 0) {
      return {
        headerRowIndex: i,
        idxDni: foundDni,
        idxMonto: foundMonto,
        idxCodigoOperacion: foundCodigo,
        idxFechaPago: foundFecha,
        idxHoraProceso: foundHora,
        idxBancoIngreso: foundBanco,
        idxEstado: foundEstado,
      };
    }
  }

  return null;
}

import * as XLSX from "xlsx";

function hasExternalSheetReference(content: string): boolean {
  return /sheet\d+\.htm/i.test(content) || /_archivos\//i.test(content);
}

function parseHtmlMatrix(
  content: string,
  config: HeaderAliasConfig,
): (string | number | null)[][] | null {
  const parser = new DOMParser();
  const doc = parser.parseFromString(content, "text/html");
  const tables = Array.from(doc.querySelectorAll("table"));

  if (tables.length === 0) return null;

  let bestMatrix: (string | number | null)[][] = [];
  let bestScore = -1;

  for (const table of tables) {
    const sheet = XLSX.utils.table_to_sheet(table);
    const matrix = XLSX.utils.sheet_to_json<(string | number | null)[]>(
      sheet,
      { header: 1, defval: "", raw: true },
    );

    const nonEmptyCells = matrix.reduce(
      (acc, row) => acc + row.filter((x) => String(x ?? "").trim() !== "").length,
      0,
    );

    const hasHeaders = !!findHeaderMatch(matrix, config);
    const score = hasHeaders ? nonEmptyCells + 10000 : nonEmptyCells;

    if (score > bestScore) {
      bestScore = score;
      bestMatrix = matrix;
    }
  }

  return bestMatrix.length > 0 ? bestMatrix : null;
}

export function readMatrix(
  buffer: ArrayBuffer,
  config: HeaderAliasConfig,
  fileName?: string,
): (string | number | null)[][] | null {
  const ext = fileName ? fileName.split(".").pop()?.toLowerCase() : "";
  const isHtmlExt = ext === "htm" || ext === "html";

  const decoder = new TextDecoder("utf-8", { fatal: false });
  const text = decoder.decode(buffer.slice(0, Math.min(buffer.byteLength, 1024 * 1024)));

  if (
    isHtmlExt ||
    text.trim().toLowerCase().startsWith("<html") ||
    text.trim().startsWith("<table") ||
    text.includes("<table")
  ) {
    const fullText = new TextDecoder("latin1").decode(buffer);
    const htmlMatrix = parseHtmlMatrix(fullText, config);
    if (htmlMatrix && htmlMatrix.length > 1) return htmlMatrix;

    // HTML exportado con referencias externas no viene autocontenido.
    if (hasExternalSheetReference(fullText)) return null;
  }

  const readOptions: XLSX.ParsingOptions[] = [
    { type: "array", raw: true },
    { type: "array", raw: true, codepage: 1252 },
    { type: "array", raw: false },
  ];

  for (const options of readOptions) {
    try {
      let workbook = XLSX.read(buffer, options);

      // Normaliza el workbook reescribiendolo como xlsx en memoria.
      const xData = XLSX.write(workbook, { type: "array", bookType: "xlsx" });
      workbook = XLSX.read(xData, { type: "array" });

      const sheetName = workbook.SheetNames[0];
      const sheet = workbook.Sheets[sheetName];
      if (!sheet) continue;

      const matrix = XLSX.utils.sheet_to_json<(string | number | null)[]>(
        sheet,
        { header: 1, defval: "", raw: true },
      );

      if (matrix.length > 1) return matrix;
    } catch {
      continue;
    }
  }

  return null;
}

function formatLocalDateTime(date: Date): string {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, "0");
  const d = String(date.getDate()).padStart(2, "0");
  const hh = String(date.getHours()).padStart(2, "0");
  const mm = String(date.getMinutes()).padStart(2, "0");
  const ss = String(date.getSeconds()).padStart(2, "0");
  return `${y}-${m}-${d} ${hh}:${mm}:${ss}`;
}

function normalizeDateToIso(value: unknown): string {
  if (value == null || String(value).trim() === "") {
    return formatLocalDateTime(new Date());
  }

  if (typeof value === "number" && Number.isFinite(value)) {
    const date = XLSX.SSF.parse_date_code(value);
    if (date) {
      return `${String(date.y).padStart(4, "0")}-${String(date.m ?? 1).padStart(2, "0")}-${String(date.d ?? 1).padStart(2, "0")} ${String(date.H ?? 0).padStart(2, "0")}:${String(date.M ?? 0).padStart(2, "0")}:${String(date.S ?? 0).padStart(2, "0")}`;
    }
  }

  const str = String(value).trim();

  const yyyymmddTime = str.match(/^(\d{4})(\d{2})(\d{2})\s+(\d{2}):(\d{2}):(\d{2})$/);
  if (yyyymmddTime) {
    return `${yyyymmddTime[1]}-${yyyymmddTime[2]}-${yyyymmddTime[3]} ${yyyymmddTime[4]}:${yyyymmddTime[5]}:${yyyymmddTime[6]}`;
  }

  const yyyy_mm_dd = str.match(/^(\d{4})[/-](\d{2})[/-](\d{2})(?:\s+(\d{2}):(\d{2})(?::(\d{2}))?)?$/);
  if (yyyy_mm_dd) {
    const year = Number(yyyy_mm_dd[1]);
    const month = Number(yyyy_mm_dd[2]);
    const day = Number(yyyy_mm_dd[3]);
    if (month >= 1 && month <= 12) {
      const hours = Number(yyyy_mm_dd[4] ?? 0);
      const minutes = Number(yyyy_mm_dd[5] ?? 0);
      const seconds = Number(yyyy_mm_dd[6] ?? 0);
      return `${String(year).padStart(4, "0")}-${String(month).padStart(2, "0")}-${String(day).padStart(2, "0")} ${String(hours).padStart(2, "0")}:${String(minutes).padStart(2, "0")}:${String(seconds).padStart(2, "0")}`;
    }
  }

  const yyyymmdd = str.match(/^(\d{4})(\d{2})(\d{2})$/);
  if (yyyymmdd) {
    return `${yyyymmdd[1]}-${yyyymmdd[2]}-${yyyymmdd[3]} 00:00:00`;
  }

  const ddmmyyyy = str.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{2,4})(?:\s+(\d{1,2}):(\d{2})(?::(\d{2}))?)?$/);
  if (ddmmyyyy) {
    const day = Number(ddmmyyyy[1]);
    const month = Number(ddmmyyyy[2]);
    const yearRaw = Number(ddmmyyyy[3]);
    const year = yearRaw < 100 ? 2000 + yearRaw : yearRaw;
    const hours = Number(ddmmyyyy[4] ?? 0);
    const minutes = Number(ddmmyyyy[5] ?? 0);
    const seconds = Number(ddmmyyyy[6] ?? 0);
    return `${String(year).padStart(4, "0")}-${String(month).padStart(2, "0")}-${String(day).padStart(2, "0")} ${String(hours).padStart(2, "0")}:${String(minutes).padStart(2, "0")}:${String(seconds).padStart(2, "0")}`;
  }

  const parsed = new Date(str);
  if (!Number.isNaN(parsed.getTime())) return formatLocalDateTime(parsed);

  return formatLocalDateTime(new Date());
}

function normalizeDateToIsoSafetyPay(value: unknown): string {
  if (value == null || String(value).trim() === "") {
    return formatLocalDateTime(new Date());
  }

  if (typeof value === "number" && Number.isFinite(value)) {
    const date = XLSX.SSF.parse_date_code(value);
    if (date) {
      return `${String(date.y).padStart(4, "0")}-${String(date.m ?? 1).padStart(2, "0")}-${String(date.d ?? 1).padStart(2, "0")} ${String(date.H ?? 0).padStart(2, "0")}:${String(date.M ?? 0).padStart(2, "0")}:${String(date.S ?? 0).padStart(2, "0")}`;
    }
  }

  const str = String(value).trim();

  if (/^\d+(?:\.\d+)?$/.test(str)) {
    const serial = Number(str);
    if (Number.isFinite(serial)) {
      const date = XLSX.SSF.parse_date_code(serial);
      if (date) {
        return `${String(date.y).padStart(4, "0")}-${String(date.m ?? 1).padStart(2, "0")}-${String(date.d ?? 1).padStart(2, "0")} ${String(date.H ?? 0).padStart(2, "0")}:${String(date.M ?? 0).padStart(2, "0")}:${String(date.S ?? 0).padStart(2, "0")}`;
      }
    }
  }

  // SafetyPay puede venir en formato MM/DD/YYYY.
  const mmddyyyy = str.match(/^(\d{1,2})[/-](\d{1,2})[/-](\d{2,4})(?:\s+(\d{1,2}):(\d{2})(?::(\d{2}))?)?$/);
  if (mmddyyyy) {
    const month = Number(mmddyyyy[1]);
    const day = Number(mmddyyyy[2]);
    const yearRaw = Number(mmddyyyy[3]);
    const year = yearRaw < 100 ? 2000 + yearRaw : yearRaw;
    const hours = Number(mmddyyyy[4] ?? 0);
    const minutes = Number(mmddyyyy[5] ?? 0);
    const seconds = Number(mmddyyyy[6] ?? 0);

    if (month >= 1 && month <= 12 && day >= 1 && day <= 31) {
      return `${String(year).padStart(4, "0")}-${String(month).padStart(2, "0")}-${String(day).padStart(2, "0")} ${String(hours).padStart(2, "0")}:${String(minutes).padStart(2, "0")}:${String(seconds).padStart(2, "0")}`;
    }
  }

  return normalizeDateToIso(value);
}

export function buildExcelRowsFromMatrix(
  matrix: (string | number | null)[][],
  match: HeaderMatch | null,
  ruta: string,
): Record<string, string>[] {
  const rows: Record<string, string>[] = [];
  if (!match) return rows;

  for (let i = match.headerRowIndex + 1; i < matrix.length; i++) {
    const row = matrix[i] ?? [];

    if (row.every((cell) => !String(cell ?? "").trim())) continue;

    const getValue = (idx: number, off = 0) =>
      row[idx + off] !== undefined && row[idx + off] !== null
        ? String(row[idx + off]).trim()
        : "";

    let offset = 0;

    // Caso especial BCP: algunos archivos vienen desplazados una columna.
    if (ruta === "/operaciones/bcp") {
      const strictDni = match.idxDni >= 0 ? getValue(match.idxDni, 0) : "";
      const strictMonto = getValue(match.idxMonto, 0);
      const strictCodigo = getValue(match.idxCodigoOperacion, 0);
      const strictFecha = getValue(match.idxFechaPago, 0);

      const shiftedDni = match.idxDni >= 0 ? getValue(match.idxDni, -1) : "";
      const shiftedMonto = getValue(match.idxMonto, -1);
      const shiftedCodigo = getValue(match.idxCodigoOperacion, -1);
      const shiftedFecha = getValue(match.idxFechaPago, -1);

      const hasStrict = !!strictDni || !!strictMonto || !!strictCodigo || !!strictFecha;
      const hasShifted = !!shiftedDni || !!shiftedMonto || !!shiftedCodigo || !!shiftedFecha;

      if (!hasStrict && hasShifted) offset = -1;
    }

    const dni = match.idxDni >= 0 ? getValue(match.idxDni, offset) : "";
    const montoRaw = getValue(match.idxMonto, offset);
    const codigoOperacion = getValue(match.idxCodigoOperacion, offset);
    const fechaPagoRaw = match.idxFechaPago >= 0 ? getValue(match.idxFechaPago, offset) : "";
    const horaProceso = match.idxHoraProceso >= 0 ? getValue(match.idxHoraProceso, offset) : "";
    const bancoIngreso = match.idxBancoIngreso >= 0 ? getValue(match.idxBancoIngreso, offset) : "";
    const estado = match.idxEstado >= 0 ? getValue(match.idxEstado, offset) : "";

    const estadoLower = estado.toLowerCase();
    if (estadoLower.includes("total de registros") || estadoLower.includes("total registros")) {
      continue;
    }

    const fechaPago =
      ruta === "/operaciones/safetypay"
        ? normalizeDateToIsoSafetyPay(fechaPagoRaw)
        : normalizeDateToIso(fechaPagoRaw);

    rows.push({
      dni,
      monto: montoRaw,
      codigo_operacion: codigoOperacion,
      fecha_pago: fechaPago,
      hora_proceso: horaProceso,
      nombre_cliente_banco_ingreso: bancoIngreso,
      estado,
    });
  }

  return rows;
}

async function handleProcessExcel({
  uploadedFiles,
  ruta,
  selectedCajaEmpresa,
  selectedCajaNombre,
  config,
  currentUser,
  getCurrentUserId,
  rpcPut,
}: {
  uploadedFiles: Array<{ file?: File }>;
  ruta: string;
  selectedCajaEmpresa: number | null;
  selectedCajaNombre: string | null;
  config: any;
  currentUser?: any;
  getCurrentUserId: () => Promise<number | null>;
  rpcPut: <T>(fn: string, params: any) => Promise<{ data?: T; error?: any }>;
}) {
  if (uploadedFiles.length === 0) {
    throw new Error("No hay archivos para procesar.");
  }

  if (!selectedCajaEmpresa) {
    throw new Error("Seleccione una empresa antes de cargar el Excel.");
  }

  const excelConfig = getExcelConfig(ruta);
  if (!excelConfig) {
    throw new Error(`No hay configuracion de columnas para: ${ruta}`);
  }

  const allRows: Record<string, string>[] = [];

  for (const file of uploadedFiles) {
    if (!file.file) continue;

    const buffer = await file.file.arrayBuffer();
    const matrix = readMatrix(buffer, excelConfig, file.file.name);

    if (!matrix || matrix.length < 2) {
      throw new Error("No se pudo leer el archivo o esta vacio.");
    }

    const match = findHeaderMatch(matrix, excelConfig);
    if (!match) {
      throw new Error("No se encontraron encabezados de columna en el archivo.");
    }

    allRows.push(...buildExcelRowsFromMatrix(matrix, match, ruta));
  }

  if (allRows.length === 0) {
    throw new Error("No se encontraron filas para procesar.");
  }

  const userId = await getCurrentUserId();

  const params = {
    p_datos_excel_json: allRows,
    p_id_codigo_servicio: config?.config?.id_codigo_servicio,
    p_nombre_codigo_servicio: config?.config?.codigo_servicio_nombre,
    p_id_comunidad: config?.config?.id_comunidad,
    p_nombre_comunidad: config?.config?.comunidad_nombre,
    p_id_tipo_deposito: config?.config?.id_tipo_deposito,
    p_nombre_tipo_deposito: config?.config?.tipo_deposito_nombre,
    p_id_empresa_ingreso: selectedCajaEmpresa,
    p_nombre_empresa_ingreso: selectedCajaNombre,
    p_id_banco: config?.config?.id_banco ?? null,
    p_id_usuario: userId,
    p_nombre_usuario: currentUser?.nombre ?? undefined,
    p_ruta_actual: ruta,
  };

  // En la web compactada usar la funcion POST compacta.
  return rpcPut("f_post_tb_operacion_proceso_excel", params);
}

8. REGLA DE ORO
====================================================

No hardcodear cabeceras dentro del componente visual.

El componente visual solo debe:
1. recibir ruta;
2. obtener configuracion por ruta;
3. leer archivo;
4. construir p_datos_excel_json;
5. llamar RPC;
6. mostrar contadores y errores.
